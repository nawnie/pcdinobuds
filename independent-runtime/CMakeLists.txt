cmake_minimum_required(VERSION 3.20)
project(yoshi_independent_runtime C)
set(CMAKE_C_STANDARD 11)
if(NOT YOSHI_RUNTIME_HEADERS)
  message(FATAL_ERROR "Set YOSHI_RUNTIME_HEADERS to the existing generated-code runtime headers")
endif()
add_library(yoshi_guest_adapter STATIC src/guest_adapter.c src/legacy_module_bridge.c src/guest_memory.c src/guest_memory_windows.c)
target_include_directories(yoshi_guest_adapter PUBLIC src "${YOSHI_RUNTIME_HEADERS}")
add_executable(yoshi_guest_adapter_test tests/guest_adapter_test.c)
target_link_libraries(yoshi_guest_adapter_test PRIVATE yoshi_guest_adapter)
enable_testing()
add_test(NAME guest_adapter COMMAND yoshi_guest_adapter_test)
add_executable(yoshi_guest_memory_test tests/guest_memory_test.c)
target_link_libraries(yoshi_guest_memory_test PRIVATE yoshi_guest_adapter)
add_test(NAME guest_memory COMMAND yoshi_guest_memory_test)
if(WIN32)
  add_executable(yoshi_guest_memory_windows_test tests/guest_memory_windows_test.c)
  target_link_libraries(yoshi_guest_memory_windows_test PRIVATE yoshi_guest_adapter)
  add_test(NAME guest_memory_windows COMMAND yoshi_guest_memory_windows_test)
endif()
